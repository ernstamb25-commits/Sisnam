1. Skema Rangkaian Hardware (Wiring per ESP32)

Rangkai sensor radar mmWave HLK-LD2410C dan modul Relay pada masing-masing board ESP32 sesuai tabel pin berikut:

Komponen Pin Komponen Pin ESP32 Keterangan
HLK-LD2410C VCC, GND, OUT 5V (VIN), GND, GPIO 22 Pin OUT memberikan logika HIGH (3.3V) saat mendeteksi keberadaan manusia
Modul Relay VCC, GND, IN 5V (VIN), GND, GPIO 23 Default kode menggunakan tipe Active LOW (RELAY_ON = LOW)

2. Cara Upload Kode .ino ke ESP32 (Arduino IDE)

  1. Instal Board ESP32: Buka Arduino IDE → masuk ke File > Preferences → tambahkan URL board ESP32 Expressif di Additional Boards Manager URLs, lalu instal package esp32 by Espressif Systems melalui menu Tools > Board > Boards Manager.
  2. Salin Kode Firmware: Buat sketch baru, hapus isinya, dan tempelkan kode sisnam_esp32.ino di bawah.
  3. Sesuaikan WiFi: Ubah nilai variabel ssid dan pass pada baris konfigurasi agar sama dengan jaringan WiFi/Hotspot tempat laptop/PC Anda terhubung.
  4. Pilih Board & Port: Hubungkan ESP32 ke PC dengan kabel USB data. Pilih Tools > Board > ESP32 Arduino > DOIT ESP32 DEVKIT V1 (atau tipe board ESP32 yang sesuai) dan pilih Port COM yang terdeteksi.
  5. Upload & Cek IP Address: Klik tombol Upload. Setelah selesai, buka Tools > Serial Monitor dan atur baud rate ke 115200. Tekan tombol EN / RST di ESP32, lalu catat alamat IP yang muncul (contoh: 192.168.1.105).
  6. Untuk Banyak Ruangan: Ulangi proses upload ke ESP32 kedua, ketiga, dst. menggunakan nama WiFi yang sama. Setiap ESP32 akan mendapatkan IP unik dari router.
sisnam_esp32.ino 
// ============================================================================
// SISNAM - Sistem Monitoring & Kontrol Kamar Mandi Pintar (ESP32)
// Sensor Kehadiran Radar mmWave HLK-LD2410C (GPIO 22) + Relay Lampu (GPIO 23)
// Kelompok 8 - Mendukung Endpoint /status & /control?mode=AUTO|ON|OFF
// ============================================================================

#include <WiFi.h>
#include <WebServer.h>

// --- KONFIGURASI WIFI ---
const char* ssid = "vanity";
const char* pass = "gatauluoa";

// --- KONFIGURASI PIN ---
const int sensorPin = 22; // Pin OUT dari HLK-LD2410C ke GPIO 22
const int relayPin  = 23; // Pin IN Relay Lampu (Silakan ubah sesuai rangkaian Anda)

// Konfigurasi Tipe Relay (Kebanyakan modul relay di pasaran aktif saat diberi logika LOW)
// Ubah menjadi HIGH jika relay Anda aktif saat diberi logika HIGH
const int RELAY_ON  = LOW;
const int RELAY_OFF = HIGH;

// --- VARIABEL STATUS SISTEM ---
// Mode: 0 = AUTO (Ikut Sensor), 1 = MANUAL ON (Paksa Nyala), 2 = MANUAL OFF (Paksa Mati)
int systemMode = 0; 
bool adaOrang = false;
bool lampuMenyala = false;

WebServer server(80);

// Fungsi untuk menyinkronkan kondisi Relay berdasarkan Mode & Sensor
void updateRelayState() {
  if (systemMode == 0) {
    // Mode Otomatis: Mengikuti pembacaan sensor radar HLK-LD2410C
    lampuMenyala = adaOrang;
  } else if (systemMode == 1) {
    // Mode Paksa ON
    lampuMenyala = true;
  } else if (systemMode == 2) {
    // Mode Paksa OFF
    lampuMenyala = false;
  }

  digitalWrite(relayPin, lampuMenyala ? RELAY_ON : RELAY_OFF);
}

// Endpoint GET /status -> Mengirim data ke Aplikasi Python / Web dalam format JSON
void handleStatus() {
  String modeStr = "AUTO";
  if (systemMode == 1) modeStr = "MANUAL_ON";
  else if (systemMode == 2) modeStr = "MANUAL_OFF";

  String json = "{";
  json += "\"presence\":" + String(adaOrang ? "true" : "false") + ",";
  json += "\"lamp\":" + String(lampuMenyala ? "true" : "false") + ",";
  json += "\"mode\":\"" + modeStr + "\",";
  json += "\"rssi\":" + String(WiFi.RSSI()) + ",";
  json += "\"ip\":\"" + WiFi.localIP().toString() + "\"";
  json += "}";

  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "application/json", json);
}

// Endpoint GET /control?mode=... -> Menerima perintah dari tombol Aplikasi Python / Web
void handleControl() {
  if (server.hasArg("mode")) {
    String reqMode = server.arg("mode");
    if (reqMode == "AUTO") {
      systemMode = 0;
      Serial.println("Mode diubah ke: OTOMATIS (Sensor)");
    } else if (reqMode == "ON") {
      systemMode = 1;
      Serial.println("Mode diubah ke: MANUAL ON (Paksa Nyala)");
    } else if (reqMode == "OFF") {
      systemMode = 2;
      Serial.println("Mode diubah ke: MANUAL OFF (Paksa Mati)");
    }
    updateRelayState();
    handleStatus(); // Kembalikan status terbaru
  } else {
    server.sendHeader("Access-Control-Allow-Origin", "*");
    server.send(400, "text/plain", "Parameter 'mode' diperlukan");
  }
}

void setup() {
  Serial.begin(115200);

  // Pin sensor menggunakan INPUT_PULLDOWN agar tidak floating
  pinMode(sensorPin, INPUT_PULLDOWN);
  
  // Set pin relay sebagai OUTPUT dan matikan di awal
  pinMode(relayPin, OUTPUT);
  digitalWrite(relayPin, RELAY_OFF);

  Serial.println("\nMenghubungkan ke WiFi: " + String(ssid));
  WiFi.begin(ssid, pass);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nBerhasil terhubung ke WiFi!");
  Serial.print("IP Address ESP32 (Masukkan ke Aplikasi Python/Web): ");
  Serial.println(WiFi.localIP());

  // Daftarkan rute Web Server
  server.on("/status", HTTP_GET, handleStatus);
  server.on("/control", HTTP_GET, handleControl);
  server.begin();
  Serial.println("HTTP Server Lokal Aktif!");
}

void loop() {
  server.handleClient();

  // Baca output digital dari pin OUT sensor HLK-LD2410C
  int bacaSensor = digitalRead(sensorPin);
  bool statusSekarang = (bacaSensor == HIGH);

  // Jika terjadi perubahan deteksi keberadaan
  if (statusSekarang != adaOrang) {
    adaOrang = statusSekarang;
    if (adaOrang) {
      Serial.println("Sensor Radar -> Ada Orang!");
    } else {
      Serial.println("Sensor Radar -> Ruangan Kosong");
    }
  }

  // Terapkan logika ke relay
  updateRelayState();
  delay(20);
}

3. Cara Menjalankan Sistem & Monitoring Multi-Ruangan

  1. Tanpa Edit Kode IP: Anda tidak perlu menulis IP ESP32 di dalam file HTML maupun di dalam web_bridge.py. Semua penambahan dan pengeditan IP dilakukan langsung dari form di halaman Dashboard Ruangan.
  2. Opsi A — Buka Langsung (sisnam.html): Klik 2x file sisnam.html di browser. Pastikan opsi Metode Akses di kanan atas disetel ke Direct HTTP (Langsung ke IP ESP32). Karena kode .ino sudah menyertakan header Access-Control-Allow-Origin: *, browser dapat langsung mengambil data ke masing-masing IP ESP32.
  3. Opsi B — Lewat Server Python (web_bridge.py): Jalankan python web_bridge.py di terminal, lalu buka http://localhost:8080. Anda dapat memilih metode akses Via Proxy (web_bridge.py) yang akan meneruskan request ke IP masing-masing ruangan secara dinamis (/status?ip=192.168.x.x).
  4. Menambahkan Ruangan: Masukkan Nama Ruangan (misal: Kamar Mandi Lt. 1), ketik IP ESP32 yang tampil di Serial Monitor, pilih mode ESP32 Live (atau Simulasi jika ingin menguji UI tanpa alat), lalu klik + Tambah Ruangan. Data daftar ruangan otomatis tersimpan di localStorage browser.
